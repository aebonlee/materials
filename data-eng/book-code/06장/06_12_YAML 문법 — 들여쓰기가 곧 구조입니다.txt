# 1. 키: 값 (콜론 뒤에 공백 하나가 필요합니다)
name: postgres

# 2. 들여쓰기로 계층을 만듭니다 — 공백 2칸이 관례, 탭은 금지
database:
  host: db
  port: 5432

# 3. 목록은 하이픈(-)
ports:
  - "5432:5432"
  - "8081:80"

# 4. 주석은 #
# 5. 포트처럼 콜론이 든 값은 늘 따옴표로 감싸는 것이 관례입니다
value: "8081:80"      # 따옴표가 없으면 22:22처럼 콜론 뒤가 60 미만인 값은 60진수 숫자(시:분처럼)로 읽힙니다(YAML 1.1)
