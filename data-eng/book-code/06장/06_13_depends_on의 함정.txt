# (후) 준비 완료까지 기다리게 하려면 healthcheck와 함께 씁니다
db:
  image: postgres:16
  healthcheck:
    test: ["CMD-SHELL", "pg_isready -U engineer"]
    interval: 5s
    retries: 5

app:
  depends_on:
    db:
      condition: service_healthy
